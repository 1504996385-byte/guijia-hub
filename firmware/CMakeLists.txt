set(APP_PATH ${CMAKE_CURRENT_LIST_DIR})
get_filename_component(APP_NAME ${APP_PATH} NAME)
file(GLOB APP_SRC CONFIGURE_DEPENDS ${APP_PATH}/src/*.c ${APP_PATH}/core/*.c)
option(HOMEHUB_EMBED_TEST_VOICES "Embed first-boot Mandarin test WAVs" OFF)
if(NOT HOMEHUB_EMBED_TEST_VOICES)
    list(REMOVE_ITEM APP_SRC ${APP_PATH}/src/homehub_seed_assets.c)
endif()
add_library(${EXAMPLE_LIB})
if(EXISTS ${APP_PATH}/.private/homehub_cloud_private.h)
    target_include_directories(${EXAMPLE_LIB} PRIVATE ${APP_PATH}/.private)
    target_compile_definitions(${EXAMPLE_LIB} PRIVATE HH_CLOUD_PRIVATE_CONFIG=1)
endif()
if(HOMEHUB_EMBED_TEST_VOICES)
    target_compile_definitions(${EXAMPLE_LIB} PRIVATE HOMEHUB_EMBED_TEST_VOICES=1)
else()
    target_compile_definitions(${EXAMPLE_LIB} PRIVATE HOMEHUB_EMBED_TEST_VOICES=0)
endif()
target_sources(${EXAMPLE_LIB} PRIVATE ${APP_SRC})
target_sources(${EXAMPLE_LIB} PRIVATE ${CMAKE_SOURCE_DIR}/src/ai_components/ai_ui/font/font_puhui_16_2.c)
target_include_directories(${EXAMPLE_LIB} PRIVATE ${APP_PATH}/src ${APP_PATH}/core)
# The pinned SDK's checked synchronous stream-upload API is declared here.
# Revalidate this internal API when updating the SDK revision.
target_include_directories(${EXAMPLE_LIB} PRIVATE ${CMAKE_SOURCE_DIR}/src/tuya_ai_service/svc_ai_agent/src)
target_compile_options(${EXAMPLE_LIB} PRIVATE -Wall -Wextra -DLV_LVGL_H_INCLUDE_SIMPLE)
